package main

// Copy this file to w05_old_writer_fixture_test.go in the immutable c8242e0
// source tree. Keep migration 18 outside migrations/ so the old binary embeds
// only v17: migrations-v18/00018_asset_classification.sql. Run this test alone
// because goose's BaseFS is process-global. No worker, scheduler, resolver,
// collector, or target request is started by this fixture.
import (
	"context"
	"testing"

	"github.com/jackc/pgx/v5/stdlib"
	"github.com/pressly/goose/v3"
)

func TestW05OldWriterKeepsNewColumns(t *testing.T) {
	s, _ := p0Store(t)
	ctx := context.Background()
	defer goose.SetBaseFS(migrations)

	version := func() int64 {
		t.Helper()
		var v int64
		if err := s.DB.QueryRow(ctx, `SELECT max(version_id) FROM (SELECT DISTINCT ON(version_id) version_id,is_applied FROM goose_db_version ORDER BY version_id,id DESC) x WHERE is_applied`).Scan(&v); err != nil {
			t.Fatal("cannot inspect isolated migration version")
		}
		return v
	}
	if version() != 17 {
		t.Fatal("old-writer fixture must start from the old source's schema 17")
	}
	if _, err := migrations.ReadFile("migrations/00018_asset_classification.sql"); err == nil {
		t.Fatal("migration 18 must not be embedded in the old source tree")
	}

	// This wrapper uses the fixture pool and its private search_path. It does not
	// replace the fixture's explicit loopback/database destination restrictions.
	sqlDB := stdlib.OpenDBFromPool(s.DB)
	defer sqlDB.Close()
	goose.SetBaseFS(nil)
	if err := goose.UpContext(ctx, sqlDB, "migrations-v18"); err != nil {
		t.Fatal("external migration 18 failed (connection details suppressed)")
	}
	goose.SetBaseFS(migrations)
	if version() != 18 {
		t.Fatal("external migration did not reach schema 18")
	}

	// Reopen with the same fixture DSN, including its generated isolated schema.
	// Do not use DATABASE_URL directly: that would discard the private search_path.
	old, err := openStore(ctx, s.DB.Config().ConnConfig.ConnString())
	if err != nil {
		t.Fatal("old v17 program cannot open schema 18 (connection details suppressed)")
	}
	defer old.DB.Close()
	if version() != 18 {
		t.Fatal("old program startup changed the migration version")
	}

	scope := insertScope(t, old, 1)
	const host = "api2.example.com"
	const manual = "人工回滚保留"
	const classification = `{"version":"subdomain-tokens-v2","mode":"manual","root":"example.com","keyword":"api","label":"api2","automatic_category":"API","base_points":10,"keyword_points":20,"alive_points":0}`
	var assetID int64
	if err := old.DB.QueryRow(ctx, `INSERT INTO assets(scope_id,host,category,score,is_alive,classification,category_override) VALUES($1,$2,$3,30,false,$4::jsonb,$3) RETURNING id`, scope.ID, host, manual, classification).Scan(&assetID); err != nil {
		t.Fatal("cannot create synthetic schema-18 asset")
	}

	run := claimedRun(t, old, scope)
	// Supply an observed HTTP response directly to the old writer. This exercises
	// its normal successful-HTTP update path without making any network request.
	if _, err := old.WriteResults(ctx, run, []Observation{{Host: host, Sources: []string{"fixture-http"}, Alive: ptr(true), Port: 443, StatusCode: 200, Title: "synthetic rollback fixture", Server: "fixture", Fingerprints: []string{}, Redirects: []string{}}}, false); err != nil {
		t.Fatal("old writer failed against schema 18 (connection details suppressed)")
	}
	closeRun(t, old, run)

	var category, override string
	var score, statusCode int
	var alive, sameClassification bool
	if err := old.DB.QueryRow(ctx, `SELECT category,score,category_override,classification=$2::jsonb,is_alive,status_code FROM assets WHERE id=$1`, assetID, classification).Scan(&category, &score, &override, &sameClassification, &alive, &statusCode); err != nil {
		t.Fatal("cannot read synthetic asset after the old writer")
	}
	if category != manual || override != manual || score != 30 || !sameClassification {
		t.Fatal("old writer changed the manual category, override, score, or classification")
	}
	if !alive || statusCode != 200 {
		t.Fatal("old writer did not execute the successful HTTP observation path")
	}
	if version() != 18 {
		t.Fatal("old writer changed the migration version")
	}
	t.Log("v17 openStore and HTTP WriteResults accepted schema 18; manual category and nullable classification columns retained; old score remains unchanged")
}
